import click
from .execution_engine import Executor, ExecutionTask, ExecutionResult

class ExecutionTask:
    def __init__(self, task_id, command):
        self.task_id = task_id
        self.command = command
        self.status = "pending"

def execute_task(task):
    """Mark the task as executed and return a result."""
    task.status = "completed"
    return ExecutionResult(task)

class ExecutionResult:
    def __init__(self, task):
        self.task = task

def execute_plan(plan):
    for task in plan:
        result = execute_task(task)
        print(f"Task {task.task_id} executed: {result.task.status}")

@click.command()
@click.option('--goal', help='The goal or plan to execute')
def main(goal):
    """CLI entry point."""
    if not goal:
        raise click.ClickException("Missing --goal option")

    # Example tasks for demonstration
    tasks = [
        ExecutionTask(task_id=1, command="echo Hello World"),
        ExecutionTask(task_id=2, command="nonexistent_command")
    ]

    executor = Executor()
    plan = [task.task_id for task in tasks]
    execute_plan(plan)
    result = execute_task(tasks[0])  # Show how to use the class
    print(f"First Task Result: {result.task.status}")

if __name__ == "__main__":
    main()

def test_execute_task():
    task = ExecutionTask(task_id=1, command="echo Hello World")
    with patch.object(ExecutionResult, 'task', new_callable=Mock) as mock_result:
        result = ExecutionResult(task)
        assert result.task == task
